  const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms));

  // Equivalent os.system('cls')
  const clear = () => { output.innerHTML = ''; };

  // Scroll automatique vers le bas
  const autoScroll = () => { terminal.scrollTop = terminal.scrollHeight; };

  // Equivalent slow_print()
  async function slowPrint(text, delay = 20, className = '') {
    const div = document.createElement('div');
    div.className = 'line ' + className;
    output.appendChild(div);
    
    // Si c'est juste un saut de ligne vide
    if (!text) { 
        div.innerHTML = '&nbsp;'; 
        return; 
    }

    for (let char of text) {
      div.textContent += char;
      autoScroll();
      await sleep(delay);
    }
  }

  // Equivalent loading_bar()
  async function loadingBar(label, duration = 1500) {
    const div = document.createElement('div');
    div.className = 'line c-dim';
    output.appendChild(div);
    div.textContent = label + " ";
    
    const steps = 30;
    const stepTime = duration / steps;

    for (let i = 0; i < steps; i++) {
      div.textContent += "█";
      autoScroll();
      await sleep(stepTime);
    }
    div.textContent += " ✔";
    div.className = 'line c-green'; // Devient vert à la fin
    await sleep(200);
  }

  // Equivalent input()
  function waitForInput() {
    return new Promise(resolve => {
      inputArea.style.display = 'block';
      inputField.value = '';
      inputField.focus();
      autoScroll();

      // Gestion touche Entrée
      const handler = (e) => {
        if (e.key === 'Enter') {
          const val = inputField.value.trim();
          inputField.removeEventListener('keydown', handler);
          inputArea.style.display = 'none';
          resolve(val);
        }
      };
      inputField.addEventListener('keydown', handler);
    });
  }
